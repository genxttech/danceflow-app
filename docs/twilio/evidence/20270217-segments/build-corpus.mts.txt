import { readFileSync, writeFileSync } from "node:fs";
import { estimateBilledSegments } from "./segments.ts";
const [s] = process.argv.slice(2);
const old = [...JSON.parse(readFileSync(`${s}/corpus.json`, "utf8")), ...JSON.parse(readFileSync(`${s}/corpus2.json`, "utf8"))].map((x: any) => x.b);
let seed = 20261114;
const next = () => { seed = (seed + 0x6d2b79f5) | 0; let t = Math.imul(seed ^ (seed >>> 15), 1 | seed); t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t; return ((t ^ (t >>> 14)) >>> 0) / 4294967296; };
const c = (...x: number[]) => String.fromCodePoint(...x);
const P = ["a","b"," ",".","\n","\r\n","€","{","é","’","中","안",c(0x1f600),c(0x2764,0xfe0f),c(0x1f44d,0x1f3fd),c(0x1f468,0x200d,0x1f469,0x200d,0x1f467),c(0x1f1fa,0x1f1f8),c(0x1f1fa),
  c(0x1f3f4,0xe0067,0xe0062,0xe0073,0xe0063,0xe0074,0xe007f),"e"+c(0x301),"नमस्ते","क्षत्रिय","مَرْحَبًا",c(0x600)+"12","สวัสดี",c(0x1100,0x1161,0x11a8),c(0x1100),c(0x200d),c(0x200c),c(0xfe0f),
  "שָׁלוֹם","ສະບາຍດີ",c(0x600,0x1f1fa,0x1f1fa,0x1f1fa),c(0x915,0x94d,0x93c,0x937),"Z"+c(0x301).repeat(80),c(0x1f1e6).repeat(9),c(0x10000),c(0xe0001)];
const fresh: string[] = [];
for (let i = 0; i < 4000; i++) { let b = ""; const t = Math.floor(next() * 900); while (b.length < t) b += P[Math.floor(next() * P.length)]; fresh.push(b); }
const all = [...old, ...fresh].map((b) => ({ b, e: estimateBilledSegments(b).segments }));
writeFileSync(`${s}/rr489/corpus_all.json`, JSON.stringify(all));
console.log("bodies", all.length, "max segments", Math.max(...all.map((x) => x.e)));
